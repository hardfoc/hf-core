54 ntc::AdcError ReadChannelCount(uint8_t channel, uint32_t* count) {
55 if (adc_ == nullptr || count == nullptr) return ntc::AdcError::ReadFailed;
56 hf_u32_t raw = 0;
57 auto err = adc_->ReadChannelCount(static_cast<hf_channel_id_t>(channel), raw);
58 if (err != hf_adc_err_t::ADC_SUCCESS) return ntc::AdcError::ReadFailed;
59 *count = raw;
60 return ntc::AdcError::Success;

67 *voltage_v = v;
68 return ntc::AdcError::Success;
