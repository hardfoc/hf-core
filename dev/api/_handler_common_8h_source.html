77inline void DelayMs(uint32_t ms) noexcept {
78#if defined(ESP_PLATFORM)
79 if (ms == 0U) {
80 return;
81 }
82 TickType_t ticks = pdMS_TO_TICKS(ms);
83 if (ticks == 0U) {
84 ticks = 1U;
85 }
86 vTaskDelay(ticks);
87#elif defined(HF_RTOS_FREERTOS) || (!defined(USE_HAL_DRIVER) && (defined(HF_RTOS_THREADX) || defined(HF_RTOS_NONE)))
88 while (ms > 0U) {
89 const uint16_t chunk =
90 (ms > 60000U) ? static_cast<uint16_t>(60000U)
91 : static_cast<uint16_t>(ms);
92 os_delay_msec(chunk);
93 ms -= chunk;
94 }
95#elif defined(USE_HAL_DRIVER)
96 HAL_Delay(ms);
97#else
98 /* Last-resort busy-wait (~1 ms @ ~200 MHz Cortex-M). Prefer RTOS/HAL. */
99 for (uint32_t m = 0; m < ms; ++m) {
100 for (uint32_t spin = 40000U; spin > 0U; --spin) {
101 __asm__ volatile("" ::: "memory");
102 }
103 }
104#endif
105}

117inline void DelayUs(uint32_t us) noexcept {
118 if (us >= 1000U) {
119 DelayMs((us + 999U) / 1000U);
120 return;
121 }
122#if defined(ESP_PLATFORM)
123 esp_rom_delay_us(us);

128 __asm__ volatile("" ::: "memory");
129 }
