99 Result<void> EnsureInitialized() noexcept {
100 const DeviceLock lock(*this);
101 if (!lock.Held()) {

104 return EnsureInitializedLocked();

109 Result<void> Deinitialize() noexcept {
110 const DeviceLock lock(*this);
111 if (!lock.Held()) {
112 return Track(Result<void>{Status::Busy, 0});
113 }
114 Driver* driver = DriverIfPresent();
115 if (driver == nullptr) {
116 return {};
117 }
118 Result<void> result{};
119 if constexpr (requires(Derived& handler, Driver& d) { handler.deinitializeLocked(d); }) {

121 }

123 return Track(result);

141 static_assert(std::is_same_v<std::invoke_result_t<Fn, Driver&>, Status>,
142 "VisitDriver: fn(Driver&) must return hf::Status");
143 const DeviceLock lock(*this);
144 if (!lock.Held()) {

187 std::invoke_result_t<Fn, Driver&> CallDriver(Fn&& fn) noexcept {
188 using R = std::invoke_result_t<Fn, Driver&>;
189 const DeviceLock lock(*this);
190 if (!lock.Held()) {
191 return Track(failure<R>(Status::Busy, 0));
192 }
193 const Result<void> init = EnsureInitializedLocked();
194 Driver* driver = DriverIfPresent();
195 if (!init.ok() || driver == nullptr) {
196 // Already counted by EnsureInitializedLocked().
197 return failure<R>(init.ok() ? Status::Fault : init.status, init.detail);
198 }
199 return Track(std::forward<Fn>(fn)(*driver));

249 Result<void> EnsureInitializedLocked() noexcept {
250 if (IsInitialized()) {
251 return {};
252 }
253 Result<void> result = self().initializeLocked();
254 if (result.ok() && !driver_.has_value()) {
255 result = {Status::Fault, 0}; // initializeLocked() must build the driver
256 }
257 if (!result.ok()) {
258 driver_.reset();
259 return Track(result);
260 }
261 initialized_.store(true, std::memory_order_release);
262 return Track(result);
263 }
