100inline std::size_t EncodeRecord(uint8_t* out, std::size_t cap, const CanFrame& f) noexcept {
101 if (!IsWellFormed(f) || cap < RecordSize(f)) {
102 return 0U;
103 }
104 uint32_t id = f.id;
105 if (f.IsExtended()) id |= kIdExtBit;
106 if (f.IsRemote()) id |= kIdRtrBit;
107 PutU32(&out[0], id);
108 out[4] = static_cast<uint8_t>(f.flags & kRecordFlagMask);
109 out[5] = f.len;
110 for (uint8_t i = 0; i < f.len; ++i) {
111 out[kRecordHeaderSize + i] = f.data[i];
112 }
113 return RecordSize(f);
114}

121template <typename OnFrame>
122DecodeError DecodeDatagram(const uint8_t* p, std::size_t n, Header* header, OnFrame&& on_frame) {
123 if (n < kHeaderSize) return DecodeError::Truncated;
124 if (p[0] != kVersion) return DecodeError::BadVersion;
125 Header h;
126 h.version = p[0];
127 h.flags = p[1];
128 h.seq = GetU16(&p[2]);
129 h.epoch = GetU32(&p[4]);
130 for (int pass = 0; pass < 2; ++pass) {
131 std::size_t off = kHeaderSize;
132 while (off < n) {
133 if (n - off < kRecordHeaderSize) return DecodeError::Truncated;
134 const uint32_t raw = GetU32(&p[off]);
135 CanFrame f;
136 f.flags = static_cast<uint8_t>(p[off + 4U] & kRecordFlagMask);
137 if ((p[off + 4U] & static_cast<uint8_t>(~kRecordFlagMask)) != 0U) return DecodeError::BadFrame;
138 f.len = p[off + 5U];
139 if (f.len > kFdMaxLen) return DecodeError::BadLength;
140 if (n - off - kRecordHeaderSize < f.len) return DecodeError::Truncated;
141 if ((raw & kIdExtBit) != 0U) f.flags |= frame_flag::kExt;
142 if ((raw & kIdRtrBit) != 0U) f.flags |= frame_flag::kRtr;
143 f.id = raw & ~(kIdExtBit | kIdRtrBit);
144 if (!IsWellFormed(f)) return DecodeError::BadFrame;
145 if (pass == 1) {
146 for (uint8_t i = 0; i < f.len; ++i) {
147 f.data[i] = p[off + kRecordHeaderSize + i];
148 }
149 on_frame(static_cast<const CanFrame&>(f));
150 }
151 off += kRecordHeaderSize + f.len;
152 }
153 }
154 if (header != nullptr) *header = h;
155 return DecodeError::None;
156}
