62 bool Attach(hf_co_driver_t* drv, uint8_t index, ICanPort& port, bool error_source) noexcept {
63 if (drv_ != nullptr || drv == nullptr) {
64 return false;
65 }
66 static constexpr hf_co_port_ops_t kOps{&SendTrampoline, &StatusTrampoline, &RecoverTrampoline};
67 const PortCaps caps = port.Caps();
68 drv_ = drv;
69 index_ = index;
70 port_ = &port;
71 backpressure_ = port.RxBackpressure();
72 if (!hf_co_driver_attach_port(drv, index, &kOps, this, caps.max_len,
73 error_source ? HF_CO_PORT_ERROR_SOURCE : 0U)) {
74 drv_ = nullptr;
75 port_ = nullptr;
76 return false;
77 }
78 port.SetTxDoneSink(&TxDoneTrampoline, this);
79 port.SetRxSink(&RxTrampoline, this);
80 return true;
81 }

84 void Detach() noexcept {
85 if (drv_ == nullptr) {
86 return;
87 }
88 port_->SetRxSink(nullptr, nullptr);
89 port_->SetTxDoneSink(nullptr, nullptr);
90 hf_co_driver_detach_port(drv_, index_);
91 drv_ = nullptr;
92 port_ = nullptr;
93 }

99 static hf_co_tx_result_t SendTrampoline(void* ctx, const hf_co_frame_t* frame) {
100 auto* self = static_cast<CoPortBinding*>(ctx);
101 CanFrame f;
102 FromC(*frame, f);
103 return static_cast<hf_co_tx_result_t>(self->port_->Send(f));
104 }
105 static bool StatusTrampoline(void* ctx, hf_co_port_status_t* out) {
106 const PortStatus s = static_cast<CoPortBinding*>(ctx)->port_->Status();
107 out->state = static_cast<uint8_t>(s.state);
108 out->tec = s.tec;
109 out->rec = s.rec;
110 out->rx_overruns = s.rx_overruns;
111 out->tx_queue_full = s.tx_queue_full;
112 out->bus_off_count = s.bus_off_count;
113 return true;

115 static bool RecoverTrampoline(void* ctx) { return static_cast<CoPortBinding*>(ctx)->port_->Recover(); }
116 static bool RxTrampoline(void* ctx, const CanFrame& frame) {
117 auto* self = static_cast<CoPortBinding*>(ctx);
118 hf_co_frame_t f;
119 ToC(frame, f);
120 return self->backpressure_ ? hf_co_driver_rx_offer(self->drv_, self->index_, &f)
121 : hf_co_driver_rx_push(self->drv_, self->index_, &f);
122 }
123 static void TxDoneTrampoline(void* ctx) {
124 auto* self = static_cast<CoPortBinding*>(ctx);
125 hf_co_driver_tx_done(self->drv_, self->index_);
126 }

131 bool backpressure_{false};
