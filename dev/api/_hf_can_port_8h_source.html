62constexpr bool IsExactFdLength(uint8_t len) noexcept {
63 return len <= 8U || len == 12U || len == 16U || len == 20U || len == 24U || len == 32U || len == 48U || len == 64U;

67constexpr bool IsWellFormed(const CanFrame& f) noexcept {
68 if ((f.flags & static_cast<uint8_t>(~frame_flag::kAll)) != 0U) return false;
69 if (f.id > (f.IsExtended() ? kMaxExtId : kMaxStdId)) return false;
70 if (f.IsFd()) return !f.IsRemote() && f.len <= kFdMaxLen && IsExactFdLength(f.len);
71 if ((f.flags & (frame_flag::kBrs | frame_flag::kEsi)) != 0U) return false;
72 return f.len <= kClassicMaxLen;
73}
